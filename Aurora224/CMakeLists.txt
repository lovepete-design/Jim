cmake_minimum_required(VERSION 3.25)
project(LarryReverbII VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
endif()
set(VST3_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vst3sdk" CACHE PATH "Official VST3 SDK checkout")
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(LarryReverbII source/Plugin.cpp)
target_include_directories(LarryReverbII PRIVATE "${VST3_SDK_ROOT}")
target_link_libraries(LarryReverbII PRIVATE sdk)
if(APPLE)
    target_sources(LarryReverbII PRIVATE source/Editor.mm)
    target_link_libraries(LarryReverbII PRIVATE "-framework Cocoa")
    set_source_files_properties(source/Editor.mm PROPERTIES COMPILE_FLAGS "-fno-objc-arc")
    add_executable(editor_tests tests/EditorTests.mm)
    target_include_directories(editor_tests PRIVATE "${VST3_SDK_ROOT}")
    target_link_libraries(editor_tests PRIVATE sdk "-framework Cocoa")
    set_source_files_properties(tests/EditorTests.mm PROPERTIES COMPILE_FLAGS "-fno-objc-arc")
    set_target_properties(LarryReverbII PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/source/Info.plist")
    smtg_target_set_bundle(LarryReverbII BUNDLE_IDENTIFIER "audio.triad.larryssexyreverb2" COMPANY_NAME "Triad Audio")
endif()
enable_testing()
add_executable(reverb_tests tests/ReverbTests.cpp)
target_include_directories(reverb_tests PRIVATE source)
add_test(NAME reverb_dsp COMMAND reverb_tests)
