cmake_minimum_required(VERSION 3.22)
project(TriadEffects VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRIAD_DSP_ONLY "Only build dependency-free DSP tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(triad_effects_dsp_tests tests/DSPTests.cpp)
    target_include_directories(triad_effects_dsp_tests PRIVATE Source)
    add_test(NAME effects_dsp COMMAND triad_effects_dsp_tests)
endif()
if(TRIAD_DSP_ONLY)
    return()
endif()
set(JUCE_DIR "" CACHE PATH "Optional local JUCE checkout")
if(JUCE_DIR)
    add_subdirectory("${JUCE_DIR}" juce)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.6.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(formats VST3 Standalone)
if(APPLE)
    list(APPEND formats AU)
endif()
function(add_triad_effect target title code identifier master category)
    juce_add_plugin(${target}
        COMPANY_NAME "Triad Audio" BUNDLE_ID "audio.triad.${identifier}"
        PLUGIN_MANUFACTURER_CODE Tria PLUGIN_CODE ${code}
        FORMATS ${formats} PRODUCT_NAME "${title}" DESCRIPTION "${title} audio processor"
        IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE
        MICROPHONE_PERMISSION_ENABLED TRUE
        MICROPHONE_PERMISSION_TEXT "${title} uses audio input for processing."
        VST3_CATEGORIES Fx ${category})
    target_sources(${target} PRIVATE Source/EffectsProcessor.cpp Source/EffectsEditor.cpp)
    target_compile_definitions(${target} PUBLIC TRIAD_MASTERING=${master} JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(${target} PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
    if(BUILD_TESTING)
        add_executable(${target}_tests tests/ProcessorTests.cpp)
        target_link_libraries(${target}_tests PRIVATE ${target})
        target_include_directories(${target}_tests PRIVATE Source $<TARGET_PROPERTY:${target},INCLUDE_DIRECTORIES>)
        target_compile_definitions(${target}_tests PRIVATE $<TARGET_PROPERTY:${target},COMPILE_DEFINITIONS>)
        add_test(NAME ${target}_processor COMMAND ${target}_tests)
    endif()
endfunction()
add_triad_effect(TriadDeEss "Triad DeEss" TrDe deess 0 Dynamics)
add_triad_effect(TriadMaster "Triad Master" TrMa master 1 Mastering)
