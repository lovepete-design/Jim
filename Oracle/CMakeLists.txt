cmake_minimum_required(VERSION 3.22)
project(TriadOracle VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRIAD_DSP_ONLY "Only build dependency-free synth tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(oracle_dsp_tests tests/DSPTests.cpp)
    target_include_directories(oracle_dsp_tests PRIVATE Source)
    add_test(NAME oracle_dsp COMMAND oracle_dsp_tests)
endif()
if(TRIAD_DSP_ONLY)
    return()
endif()
set(JUCE_DIR "" CACHE PATH "Optional JUCE checkout")
if(JUCE_DIR)
    add_subdirectory("${JUCE_DIR}" juce)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.6.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(formats VST3 Standalone)
if(APPLE)
    list(APPEND formats AU)
endif()
juce_add_plugin(TriadOracle COMPANY_NAME "Triad Audio" BUNDLE_ID "audio.triad.oracle"
    PLUGIN_MANUFACTURER_CODE Tria PLUGIN_CODE TrOr FORMATS ${formats}
    PRODUCT_NAME "Triad Oracle" DESCRIPTION "Five-voice analogue-inspired polysynth"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth)
target_sources(TriadOracle PRIVATE Source/SynthProcessor.cpp Source/SynthEditor.cpp)
target_compile_definitions(TriadOracle PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TriadOracle PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(BUILD_TESTING)
    add_executable(oracle_processor_tests tests/ProcessorTests.cpp)
    target_link_libraries(oracle_processor_tests PRIVATE TriadOracle)
    target_include_directories(oracle_processor_tests PRIVATE Source $<TARGET_PROPERTY:TriadOracle,INCLUDE_DIRECTORIES>)
    target_compile_definitions(oracle_processor_tests PRIVATE $<TARGET_PROPERTY:TriadOracle,COMPILE_DEFINITIONS>)
    add_test(NAME oracle_processor COMMAND oracle_processor_tests Triad-Oracle-Preview.png)
endif()
