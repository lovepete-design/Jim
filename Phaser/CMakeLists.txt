cmake_minimum_required(VERSION 3.22)
project(TriadPhase VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRIAD_DSP_ONLY "Only build dependency-free phaser tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(triad_phase_dsp_tests tests/DSPTests.cpp)
    target_include_directories(triad_phase_dsp_tests PRIVATE Source)
    add_test(NAME phase_dsp COMMAND triad_phase_dsp_tests)
endif()
if(TRIAD_DSP_ONLY)
    return()
endif()
set(JUCE_DIR "" CACHE PATH "Optional JUCE checkout")
if(JUCE_DIR)
    add_subdirectory("${JUCE_DIR}" juce)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.6.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
set(formats VST3 Standalone)
if(APPLE)
    list(APPEND formats AU)
endif()
juce_add_plugin(TriadPhase COMPANY_NAME "Triad Audio" BUNDLE_ID "audio.triad.phase"
    PLUGIN_MANUFACTURER_CODE Tria PLUGIN_CODE TrPh FORMATS ${formats}
    PRODUCT_NAME "Triad Phase" DESCRIPTION "Vintage-inspired four-stage and two-stage phaser"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Triad Phase uses audio input for processing."
    VST3_CATEGORIES Fx Modulation)
target_sources(TriadPhase PRIVATE Source/PhaserProcessor.cpp Source/PhaserEditor.cpp)
target_compile_definitions(TriadPhase PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TriadPhase PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(BUILD_TESTING)
    add_executable(triad_phase_processor_tests tests/ProcessorTests.cpp)
    target_link_libraries(triad_phase_processor_tests PRIVATE TriadPhase)
    target_include_directories(triad_phase_processor_tests PRIVATE Source $<TARGET_PROPERTY:TriadPhase,INCLUDE_DIRECTORIES>)
    target_compile_definitions(triad_phase_processor_tests PRIVATE $<TARGET_PROPERTY:TriadPhase,COMPILE_DEFINITIONS>)
    add_test(NAME phase_processor COMMAND triad_phase_processor_tests Triad-Phase-Preview.png)
endif()
